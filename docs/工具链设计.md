# Agent工作流程简图
```text
                     Agent收到用户描述和音频文件   
                               ↓
                          音乐特征分析→得到用于分析改编的文本信息
                               ↓
                           MIDI生成→得到MIDI中间产物1
                               ↓
                            改编-修改MIDI文件→得到MIDI中间产物2
                               ↓
                          渲染→得到音频文件
                               ↓
                      将结果（关于改编的文字描述）和音频文件返回给用户
```
# 工具链
```text
Librosa-提取特征
↓
Omnizart-生成MIDI
↓
maidi以及其他Python丰富的MIDI处理库-改编MIDI
↓
FluidSynth-MIDI转音频
```

## Librosa 提取音频特征
Librosa是Python生态中最全面的音频与音乐分析开源工具库，涵盖从底层信号处理到高层音乐特征提取的全流程。Librosa基于NumPy/SciPy构建，遵循命令式Python API设计，安装后在Python环境中以函数库的方式直接调用。
### 调用示例
```python
import librosa
import numpy as np

def extract_audio_features(audio_path):
    """
    使用 Librosa 从音频文件中提取多种特征
    """
    # 加载音频文件
    y, sr = librosa.load(audio_path, sr=None)

    # 1. 时域特征
    # 音高相关：Mel 频谱
    mel_spec = librosa.feature.melspectrogram(y=y, sr=sr, n_mels=128)
    mel_spec_db = librosa.power_to_db(mel_spec, ref=np.max)

    # 2. 节奏与节拍
    tempo, beat_frames = librosa.beat.beat_track(y=y, sr=sr)

    # 3. 频域特征
    # MFCC（梅尔频率倒谱系数）
    mfccs = librosa.feature.mfcc(y=y, sr=sr, n_mfcc=13)
    # 频谱质心
    spectral_centroids = librosa.feature.spectral_centroid(y=y, sr=sr)[0]
    # 色度特征
    chroma = librosa.feature.chroma_stft(y=y, sr=sr)

    # 4. 调式与和弦（简化检测，实际更复杂，需结合其他库）
    # 此处仅展示用 Chroma 特征进行简单调式归类提示
    # 更精确的调式/和弦识别建议使用专门的模型如 Essentia

    # 5. 结构划分：节拍位置检测
    onset_frames = librosa.onset.onset_detect(y=y, sr=sr)
    onset_times = librosa.frames_to_time(onset_frames, sr=sr)

    print(f"音频时长: {librosa.get_duration(y=y, sr=sr):.2f} 秒")
    print(f"估计的 BPM: {tempo:.2f}")
    print(f"MFCC 形状: {mfccs.shape}")
    print(f"第一个节拍起始时间: {onset_times[0] if len(onset_times)>0 else '无'}")

    return {
        'duration': librosa.get_duration(y=y, sr=sr),
        'tempo': tempo,
        'mfcc': mfccs,
        'chroma': chroma,
        'spectral_centroids': spectral_centroids,
        'onset_times': onset_times
    }

# 使用示例
if __name__ == "__main__":
    features = extract_audio_features("your_audio.wav")
```
## Omnizart 生成MIDI中间产物
Omnizart是一个开源的Python自动音乐转录工具，能够将复调音乐音频转录为MIDI、和弦序列及节拍等符号化表示。它支持音高乐器、声乐旋律、和弦行进、打击乐事件及节拍的全方位转录。其代码仓库也提供了Python接口供程序化调用。
### 调用示例
```python
from omnizart.music import app as music_app
from omnizart.chord import app as chord_app
from omnizart.drum import app as drum_app
import os

def transcribe_audio_to_midi(audio_path, model_type='music', output_dir='./output'):
    """
    使用 Omnizart 将音频转录为 MIDI。
    参数 model_type: 'music'（全乐器）, 'chord'（和弦）, 'drum'（鼓）
    """
    if not os.path.exists(output_dir):
        os.makedirs(output_dir)

    if model_type == 'music':
        # 转录全乐器（最常用）
        music_app.transcribe(audio_path, model_type='music', output=output_dir)
    elif model_type == 'chord':
        chord_app.transcribe(audio_path, output=output_dir)
    elif model_type == 'drum':
        drum_app.transcribe(audio_path, output=output_dir)
    else:
        raise ValueError("model_type 必须是 'music', 'chord' 或 'drum'")

    # 自动查找生成的 MIDI 文件（依据 Omnizart 命名规则）
    for f in os.listdir(output_dir):
        if f.endswith('.mid'):
            midi_path = os.path.join(output_dir, f)
            print(f"MIDI 已生成: {midi_path}")
            return midi_path
    return None

if __name__ == "__main__":
    # 转录完整音乐为 MIDI
    transcribe_audio_to_midi("your_audio.wav", model_type='music')

    # 或者只转录和弦进行
    # transcribe_audio_to_midi("your_audio.wav", model_type='chord')
```
## maidi MIDI改编与编辑
maidi是一个面向符号化音乐AI开发的Python库，主要用于MIDI文件的加载、保存、编辑、合并与分析。其核心对象`MidiScore`以“音轨×小节”的二维结构操作MIDI数据，支持调用`score.add_instrument()`添加音轨、整体移调、速度调整等改编操作，并内置了和弦提取、自动标记（密度、复音数等）等分析功能。
### 调用示例
```python
from maidi import MidiScore, instrument
import numpy as np

def edit_midi_with_maidi(input_midi_path, output_midi_path):
    """
    使用 maidi 进行 MIDI 文件的基础编辑和改编
    """
    # 加载 MIDI 文件
    score = MidiScore.from_midi(input_midi_path)
    print(f"原始小节数: {score.nb_bars}")
    print(f"原始速度: {score.tempo}")

    # 1. 改编：添加一个新音轨（例如添加一把原声吉他）
    score = score.add_instrument(instrument.ACOUSTIC_GRAND_PIANO)

    # 2. 改编：调整速度
    # 注意：maidi 中速度是作为 MidiScore 的属性直接修改
    score.tempo = 130  # 速度提升到 130 BPM

    # 3. 改编：整体移调（例如升高 2 个半音）
    # 这里需要对所有音符进行操作，maidi 提供了类似 numpy 的索引方式
    # 假设我们要将整个 score 移调 +2 个半音
    for track_idx in range(len(score.tracks)):
        for bar_idx in range(score.nb_bars):
            notes = score[track_idx, bar_idx]  # 获取该音轨该小节的音符数据
            for note in notes:
                note['pitch'] += 2  # 移调操作

    # 4. 改编：合并另一个 MIDI 文件（如果需要，展示如何横向合并）
    # score2 = MidiScore.from_midi("another_song.mid")
    # merged_score = score.concatenate(score2, axis=1)

    # 5. 保存改编后的 MIDI
    score.write(output_midi_path)
    print(f"MIDI 已保存至: {output_midi_path}")
    print(f"修改后小节数: {score.nb_bars}")
    print(f"修改后速度: {score.tempo}")

    # 6. 可选：进行自动 MIDI 标记（获取和弦、时间签名等）
    from maidi.analysis import tags_providers
    from maidi import ScoreTagger

    tagger = ScoreTagger([
        tags_providers.DensityTagsProvider(),
        tags_providers.MinMaxPolyphonyTagsProvider(),
    ])
    tags = tagger.tag_score(score)
    print("自动标记信息示例:", tags)

if __name__ == "__main__":
    edit_midi_with_maidi("input.mid", "output_edited.mid")
```
## FluidSynth MIDI转音频
FluidSynth是一个基于SoundFont 2规范的实时软件MIDI合成器，能够将MIDI指令或MIDI文件渲染为高质量音频。其支持实时播放和离线生成两种模式，配合`.sf2`音色库文件，可以向合成器发送`noteon`/`noteoff`事件，或通过`get_samples()`获取音频数据进行保存。`pyfluidsynth`库提供了Python接口，安装后通过`import fluidsynth`即可调用。
### 调用示例
```python
import fluidsynth
import numpy as np
import soundfile as sf
import time

def midi_to_audio(midi_path, sf2_path, output_wav_path, duration_seconds=30):
    """
    使用 pyfluidsynth 将 MIDI 文件转为音频（需提供 SoundFont .sf2 文件）
    """
    # 初始化合成器
    fs = fluidsynth.Synth()
    fs.start(driver="alsa")  # Linux 下常用 "alsa"，Windows 可用 "dsound" 或 "portaudio"

    # 加载 SoundFont
    sfid = fs.sfload(sf2_path)

    # 加载 MIDI 文件
    # pyfluidsynth 本身不解析 MIDI 文件，需要借助其他库（如 mido 或 pretty_midi）
    # 这里我们展示直接编程播放示例，如果需要从 MIDI 文件播放，可使用 midi 库解析后发送 note on/off

    # 示例：播放一个简单的 C 大调和弦 2 秒
    # 设置通道 0 使用 SoundFont 的第一个预设
    fs.program_select(0, sfid, 0, 0)

    # 播放音符
    fs.noteon(0, 60, 80)  # C4
    fs.noteon(0, 64, 80)  # E4
    fs.noteon(0, 67, 80)  # G4
    time.sleep(2)
    fs.noteoff(0, 60)
    fs.noteoff(0, 64)
    fs.noteoff(0, 67)
    time.sleep(0.5)

    # 也可以直接生成音频数据到 numpy 数组而不播放
    # 创建无音频驱动的 Synth，手动获取样本
    fs2 = fluidsynth.Synth()
    # 加载 SoundFont...
    sfid2 = fs2.sfload(sf2_path)
    fs2.program_select(0, sfid2, 0, 0)
    fs2.noteon(0, 60, 80)
    # 获取 2 秒的音频数据 (44100 Hz * 2 通道)
    audio_data = fs2.get_samples(44100 * 2)  # 返回 numpy 数组，立体声交错
    fs2.delete()

    # 保存为 WAV 文件
    # get_samples 返回的是交错立体声，需要 reshape 为 (samples, channels)
    if audio_data is not None and len(audio_data) > 0:
        # 确保音频数据是浮点数格式并将其转成 int16 以节省空间
        audio_int16 = (audio_data * 32767).astype(np.int16)
        # 重新排列成 (samples, 2) 的形状
        audio_2d = audio_int16.reshape(-1, 2)
        sf.write(output_wav_path, audio_2d, 44100, subtype='PCM_16')
        print(f"音频已保存至: {output_wav_path}")

    fs.delete()

if __name__ == "__main__":
    midi_to_audio("input.mid", "FluidR3_GM.sf2", "output.wav")
```
